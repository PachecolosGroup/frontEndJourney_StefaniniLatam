<script>
document.addEventListener('DOMContentLoaded', () => {
    const tabs = document.querySelectorAll('.stefa-tabbed__tab-item'); // Selecciona las pestañas
    const tabContents = document.querySelectorAll('.stefa-tabbed__parent'); // Selecciona el contenido de las pestañas

    // Añade un evento de clic a cada pestaña
    tabs.forEach((tab) => {
        tab.addEventListener('click', () => {
            const targetTab = tab.getAttribute('data-item'); // Obtiene el valor del atributo data-item

            // Quitar la clase activa de todas las pestañas
            tabs.forEach((t) => t.classList.remove('lt__tab-item--active'));

            // Agregar la clase activa a la pestaña clicada
            tab.classList.add('lt__tab-item--active');

            // Quitar la clase activa de todos los contenidos
            tabContents.forEach((content) => content.classList.remove('lt__parent--active'));

            // Agregar la clase activa al contenido correspondiente
            const activeContent = document.querySelector(`.stefa-tabbed__parent[tab-id="${targetTab}"]`);
            if (activeContent) {
                activeContent.classList.add('lt__parent--active');
            }
        });
    });
});

 
 </script>
